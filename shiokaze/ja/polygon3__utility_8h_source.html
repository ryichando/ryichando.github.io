54  static void transform( std::vector<vec3d> &vertices, vec3d origin=vec3d(), double scale=1.0, int axis=0, double rotation=0 ) {
55  //
56  // Normalize
57  size_t nvertices (vertices.size());
58  vec3d min_v;
59  vec3d max_v;
60  compute_AABB(vertices,min_v,max_v);
61  //
62  for( size_t n=0; n<nvertices; n++ ) {
63  for( int dim : DIMS3 ) {
64  min_v[dim] = std::min(vertices[n][dim],min_v[dim]);
65  max_v[dim] = std::max(vertices[n][dim],max_v[dim]);
66  }
67  }
68  double norm_s = max_v[0]-min_v[0];
69  if( norm_s ) {
70  for( size_t n=0; n<nvertices; n++ ) {
71  vertices[n] = scale * (vertices[n]-min_v) / norm_s + origin - 0.5 * scale * vec3d(1.0,0.0,1.0);
72  }
73  }
74  //
75  // Rotation
76  if( rotation ) {
77  vec3d min_v(1e18,1e18,1e18);
78  vec3d max_v = -1.0 * min_v;
79  for( size_t n=0; n<nvertices; n++ ) {
80  for( uint dim=0; dim<DIM3; dim++ ) {
81  min_v[dim] = std::min(vertices[n][dim],min_v[dim]);
82  max_v[dim] = std::max(vertices[n][dim],max_v[dim]);
83  }
84  }
85  vec3d center = 0.5 * (max_v+min_v);
86  for( size_t n=0; n<nvertices; n++ ) {
87  //
88  vec3d p = vertices[n];
89  p = p-center;
90  double theta = rotation / 180.0 * 3.1415;
91  if( axis == 0 ) {
92  p = vec3d(p[0],cos(theta)*p[1]-sin(theta)*p[2],sin(theta)*p[1]+cos(theta)*p[2]);
93  } else if( axis == 1 ) {
94  p = vec3d(cos(theta)*p[0]-sin(theta)*p[2],p[1],sin(theta)*p[0]+cos(theta)*p[2]);
95  } else if( axis == 2 ) {
96  p = vec3d(p[0],cos(theta)*p[1]-sin(theta)*p[2],sin(theta)*p[1]+cos(theta)*p[2]);
97  }
98  p = p+center;
99  vertices[n] = p;
100  }
101  }
102  }
113  template <class N> static vec3d get_center_of_gravity( std::vector<vec3d> &vertices, std::vector<std::vector<N> > &faces ) {
114  //
115  //http://stackoverflow.com/questions/2083771/a-method-to-calculate-the-centre-of-mass-from-a-stl-stereo-lithography-file
116  //
117  vec3d center_of_gravity;
118  double totalVolume(0.0), currentVolume(0.0);
119  for( const auto &triangle : faces ) {
120  //
121  assert(triangle.size() == 3);
122  vec3d v[3] = { vertices[triangle[0]], vertices[triangle[1]], vertices[triangle[2]] };
123  totalVolume += currentVolume = (
124  + v[0][0]*v[1][1]*v[2][2]
125  - v[0][0]*v[2][1]*v[1][2]
126  - v[1][0]*v[0][1]*v[2][2]
127  + v[1][0]*v[2][1]*v[0][2]
128  + v[2][0]*v[0][1]*v[1][2]
129  - v[2][0]*v[1][1]*v[0][2]) / 6.0;
130  center_of_gravity[0] += ((v[0][0]+v[1][0]+v[2][0])/4.0) * currentVolume;
131  center_of_gravity[1] += ((v[0][1]+v[1][1]+v[2][1])/4.0) * currentVolume;
132  center_of_gravity[2] += ((v[0][2]+v[1][2]+v[2][2])/4.0) * currentVolume;
133  }
134  center_of_gravity = center_of_gravity / totalVolume;
135  return center_of_gravity;
136  }
147  static void compute_AABB( std::vector<vec3d> &vertices, vec3d &corner0, vec3d &corner1 ) {
148  //
149  corner0 = vec3d(1e18,1e18,1e18);
150  corner1 = -1.0 * corner0;
151  //
152  for( size_t n=0; n<vertices.size(); n++ ) {
153  for( int dim : DIMS3 ) {
154  corner0[dim] = std::min(vertices[n][dim],corner0[dim]);
155  corner1[dim] = std::max(vertices[n][dim],corner1[dim]);
156  }
157  }
158  }
159 };
