52  static void add_container_wall( rigidworld2_interface *world, const rg2::attribution2 &attribute, const vec2d &p0, const vec2d& p1 ) {
53  //
54  vec2d center = 0.5 * (p0+p1);
55  vec2d hw = p1-center;
56  //
57  rg2::velocity2 wall_velocity = { {0.0,0.0}, 0.0 };

63  std::vector<vec2d> top = {{-hw[0],hw[1]},{hw[0],hw[1]}};
64  //

66  world->add_rigidbody(wall_polyshaps,attribute,wall_position,wall_velocity);

78  static vec2d get_velocity( const rigidbody2_interface *rigidbody, const vec2d &p ) {
79  vec2d r = p-rigidbody->get_position().center;
80  rigidbody2_interface::velocity2 u = rigidbody->get_velocity();
81  double a = u.angular_velocity;
82  return u.center_velocity+a*vec2d(-r[1],r[0]);

95  static bool getInverseOpenGLMatrix( const rigidbody2_interface *rigidbody, double inv[9] ) {
96  //
97  Real m[9];
98  rigidbody->getOpenGLMatrix(m);
99  //
100  double det = m[0]*(m[1+3*1]*m[2+3*2]-m[1+3*2]*m[2+3*1])
101  -m[1+3*0]*(m[0+3*1]*m[2+3*2]-m[2+3*1]*m[0+3*2])
102  +m[2+3*0]*(m[0+3*1]*m[1+3*2]-m[1+3*1]*m[0+3*2]);
103  //
104  if( ! det ) return false;
105  double invdet = 1.0 / det;
106  double A[9];
107  //
108  inv[0+3*0] = (m[1+3*1] * m[2+3*2] - m[2+3*1] * m[1+3*2]) * invdet;
109  inv[0+3*1] = (m[0+3*2] * m[2+3*1] - m[0+3*1] * m[2+3*2]) * invdet;
110  inv[0+3*2] = (m[0+3*1] * m[1+3*2] - m[0+3*2] * m[1+3*1]) * invdet;
111  inv[1+3*0] = (m[1+3*2] * m[2+3*0] - m[1+3*0] * m[2+3*2]) * invdet;
112  inv[1+3*1] = (m[0+3*0] * m[2+3*2] - m[0+3*2] * m[2+3*0]) * invdet;
113  inv[1+3*2] = (m[1+3*0] * m[0+3*2] - m[0+3*0] * m[1+3*2]) * invdet;
114  inv[2+3*0] = (m[1+3*0] * m[2+3*1] - m[2+3*0] * m[1+3*1]) * invdet;
115  inv[2+3*1] = (m[2+3*0] * m[0+3*1] - m[0+3*0] * m[2+3*1]) * invdet;
116  inv[2+3*2] = (m[0+3*0] * m[1+3*1] - m[1+3*0] * m[0+3*1]) * invdet;
117  return true;
118  }
129  static vec2d get_local_position( double inv[9], const vec2d &p ) {
130  //
131  vec2d result;
132  double b[] = { p[0], p[1], 1.0 };
133  for( int i=0; i<3; ++i ) for( int j=0; j<3; ++j ) {
134  result[j] += inv[i+3*j] * b[j];
135  }
136  return result;
137  }
