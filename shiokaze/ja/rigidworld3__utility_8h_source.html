53  static void add_container_wall( rigidworld3_interface *world, const rg3::attribution3 &attribute, const vec3d &p0, const vec3d& p1 ) {
54  //
55  vec3d center = 0.5 * (p0+p1);
56  vec3d hw = p1-center;
57  //
58  rg3::velocity3 wall_velocity = { vec3d(), vec3d() };
59  rg3::position3 wall_position = { center, vec3d(), 0.0 };
60  //
61  rg3::polyshape3 bottom;
62  bottom.vertices = {vec3d(-hw[0],-hw[1],-hw[2]),vec3d(hw[0],-hw[1],-hw[2]),vec3d(hw[0],-hw[1],hw[2]),vec3d(-hw[0],-hw[1],hw[2])};
63  bottom.faces = {{0,1,2},{0,2,3}};
64  bottom.type = rg3::MESH;
65  //
66  rg3::polyshape3 top;
67  top.vertices = {vec3d(-hw[0],hw[1],-hw[2]),vec3d(hw[0],hw[1],-hw[2]),vec3d(hw[0],hw[1],hw[2]),vec3d(-hw[0],hw[1],hw[2])};
68  top.faces = {{0,1,2},{0,2,3}};
69  top.type = rg3::MESH;
70  //
71  rg3::polyshape3 right;
72  right.vertices = {vec3d(hw[0],-hw[1],-hw[2]),vec3d(hw[0],-hw[1],hw[2]),vec3d(hw[0],hw[1],hw[2]),vec3d(hw[0],hw[1],-hw[2])};
73  right.faces = {{0,1,2},{0,2,3}};
74  right.type = rg3::MESH;
75  //
76  rg3::polyshape3 left;
77  left.vertices = {vec3d(-hw[0],-hw[1],-hw[2]),vec3d(-hw[0],-hw[1],hw[2]),vec3d(-hw[0],hw[1],hw[2]),vec3d(-hw[0],hw[1],-hw[2])};

103  vec3d get_velocity( const rigidbody3_interface *rigidbody, const vec3d &p ) const {
104  vec3d r = p-rigidbody->get_position().center;
105  rigidbody3_interface::velocity3 u = rigidbody->get_velocity();
106  vec3d a = u.angular_velocity;
107  return a ^ r;

120  bool getInverseOpenGLMatrix( const rigidbody3_interface *rigidbody, double invOut[16] ) const {
121  //
122  Real m[16];
123  rigidbody->getOpenGLMatrix(m);
124  //
125  double inv[16], det; int i;
126  inv[0] = m[5]*m[10]*m[15] - m[5]*m[11]*m[14] - m[9]*m[6]*m[15]
127  + m[9]*m[7]*m[14] + m[13]*m[6]*m[11] - m[13]*m[7]*m[10];
128  inv[4] = -m[4]*m[10]*m[15] + m[4]*m[11]*m[14] + m[8]*m[6]*m[15]
129  - m[8]*m[7]*m[14] - m[12]*m[6]*m[11] + m[12]*m[7]*m[10];
130  inv[8] = m[4]*m[9]*m[15] - m[4]*m[11]*m[13] - m[8]*m[5]*m[15]
131  + m[8]*m[7]*m[13] + m[12]*m[5]*m[11] - m[12]*m[7]*m[9];
132  inv[12] = -m[4]*m[9]*m[14] + m[4]*m[10]*m[13] + m[8]*m[5]*m[14]
133  - m[8]*m[6]*m[13] - m[12]*m[5]*m[10] + m[12]*m[6]*m[9];
134  inv[1] = -m[1]*m[10]*m[15] + m[1]*m[11]*m[14] + m[9]*m[2]*m[15]
135  - m[9]*m[3]*m[14] - m[13]*m[2]*m[11] + m[13]*m[3]*m[10];
136  inv[5] = m[0]*m[10]*m[15] - m[0]*m[11]*m[14] - m[8]*m[2]*m[15]
137  + m[8]*m[3]*m[14] + m[12]*m[2]*m[11] - m[12]*m[3]*m[10];
138  inv[9] = -m[0]*m[9]*m[15] + m[0]*m[11]*m[13] + m[8]*m[1]*m[15]
139  - m[8]*m[3]*m[13] - m[12]*m[1]*m[11] + m[12]*m[3]*m[9];
140  inv[13] = m[0]*m[9]*m[14] - m[0]*m[10]*m[13] - m[8]*m[1]*m[14]
141  + m[8]*m[2]*m[13] + m[12]*m[1]*m[10] - m[12]*m[2]*m[9];
142  inv[2] = m[1]*m[6]*m[15] - m[1]*m[7]*m[14] - m[5]*m[2]*m[15]
143  + m[5]*m[3]*m[14] + m[13]*m[2]*m[7] - m[13]*m[3]*m[6];
144  inv[6] = -m[0]*m[6]*m[15] + m[0]*m[7]*m[14] + m[4]*m[2]*m[15]
145  - m[4]*m[3]*m[14] - m[12]*m[2]*m[7] + m[12]*m[3]*m[6];
146  inv[10] = m[0]*m[5]*m[15] - m[0]*m[7]*m[13] - m[4]*m[1]*m[15]
147  + m[4]*m[3]*m[13] + m[12]*m[1]*m[7] - m[12]*m[3]*m[5];
148  inv[14] = -m[0]*m[5]*m[14] + m[0]*m[6]*m[13] + m[4]*m[1]*m[14]
149  - m[4]*m[2]*m[13] - m[12]*m[1]*m[6] + m[12]*m[2]*m[5];
150  inv[3] = -m[1]*m[6]*m[11] + m[1]*m[7]*m[10] + m[5]*m[2]*m[11]
151  - m[5]*m[3]*m[10] - m[9]*m[2]*m[7] + m[9]*m[3]*m[6];
152  inv[7] = m[0]*m[6]*m[11] - m[0]*m[7]*m[10] - m[4]*m[2]*m[11]
153  + m[4]*m[3]*m[10] + m[8]*m[2]*m[7] - m[8]*m[3]*m[6];
154  inv[11] = -m[0]*m[5]*m[11] + m[0]*m[7]*m[9] + m[4]*m[1]*m[11]
155  - m[4]*m[3]*m[9] - m[8]*m[1]*m[7] + m[8]*m[3]*m[5];
156  inv[15] = m[0]*m[5]*m[10] - m[0]*m[6]*m[9] - m[4]*m[1]*m[10]
157  + m[4]*m[2]*m[9] + m[8]*m[1]*m[6] - m[8]*m[2]*m[5];
158  det = m[0]*inv[0] + m[1]*inv[4] + m[2]*inv[8] + m[3]*inv[12];
159  if (det == 0) return false;
160  det = 1.0 / det;
161  for (i = 0; i < 16; i++) invOut[i] = inv[i] * det;
162  return true;
163  }
174  vec3d get_local_position( double inv[16], const vec3d &p ) const {
175  //
176  vec3d result;
177  double b[] = { p[0], p[1], p[2], 1.0 };
178  for( int i=0; i<4; ++i ) for( int j=0; j<4; ++j ) {
179  result[j] += inv[i+4*j] * b[j];
180  }
181  return result;
182  }
