179  static double get_area( const double isosurf[2][2] ) {
180  //
181  static const int quads[][2] = { {0,0}, {1,0}, {1,1}, {0,1} };
182  auto calcMarchingPoints = [&]( const double values[4], double p[8][2], int &pnum ) {
183  pnum = 0;
184  for( int n=0; n<4; n++ ) {
185  if( values[n] < 0.0 ) {
186  p[pnum][0] = quads[n][0];
187  p[pnum][1] = quads[n][1];
188  pnum ++;
189  }
190  if( values[n] * values[(n+1)%4] < 0 ) {
191  double y0 = values[n];
192  double y1 = values[(n+1)%4];
193  if( y0-y1 ) {
194  double a = y0/(y0-y1);
195  int p0[2] = { quads[n][0], quads[n][1] };
196  int p1[2] = { quads[(n+1)%4][0], quads[(n+1)%4][1] };
197  p[pnum][0] = (1.0-a)*p0[0]+a*p1[0];
198  p[pnum][1] = (1.0-a)*p0[1]+a*p1[1];
199  pnum ++;
200  }
201  }
202  }
203  };
204  //
205  double p[8][2], v[4];
206  for( int n=0; n<4; ++n ) v[n] = isosurf[quads[n][0]][quads[n][1]];
207  int pnum;
208  calcMarchingPoints(v,p,pnum);
209  double sum (0.0);
210  for( int m=0; m<pnum; m++ ) {
211  sum += p[m][0]*p[(m+1)%pnum][1]-p[m][1]*p[(m+1)%pnum][0];
212  }
213  return 0.5 * sum;
214  }
225  template <class N> static double compute_area( const std::vector<vec3d> &vertices, const std::vector<std::vector<N> > &faces, std::function<bool(const vec3d &p)> test_func ) {
226  //
227  double area (0.0);
228  for( const auto &f : faces ) {
229  vec3d avg;
230  for( const auto &i : f ) { avg += vertices[i]; }
231  avg /= f.size();
232  if( test_func(avg)) {
233  const vec3d &v0 = vertices[f[0]];
234  for( int n=1; n<f.size()-1; ++n ) {
235  const vec3d &v1 = vertices[f[n]];
236  const vec3d &v2 = vertices[f[n+1]];
237  area += 0.5 * ((v1-v0)^(v2-v0)).len();
238  }
239  }
240  }
241  return area;
242  }
