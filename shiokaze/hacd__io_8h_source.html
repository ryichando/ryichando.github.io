59  static std::vector<hacd_io::convex_object> read_hacd ( std::string path ) {
60  //
61  FILE *fp = fopen(path.c_str(),"rb");
62  assert(fp);
63  //
64  unsigned version;
65  assert(fread(&version,1,sizeof(unsigned),fp));
66  assert( version == 0x0001 );
67  //
68  unsigned num_objects;
69  assert(fread(&num_objects,1,sizeof(unsigned),fp));
70  std::vector<hacd_io::convex_object> objects;
71  objects.resize(num_objects);
72  //
73  for( unsigned n=0; n<num_objects; ++n ) {
74  //
75  auto &obj = objects[n];
76  auto &vertices = obj.vertices;
77  auto &faces = obj.faces;
78  //
79  unsigned num_vertices;
80  unsigned num_faces;
81  //
82  assert(fread(&num_vertices,1,sizeof(unsigned),fp));
83  assert(fread(&num_faces,1,sizeof(unsigned),fp));
84  //
85  vertices.resize(num_vertices);
86  faces.resize(num_faces);
87  //
88  for( size_t k=0; k<num_vertices; k++ ) {
89  assert(fread(&vertices[k],1,sizeof(vec3d),fp));
90  }
91  //
92  for( size_t k=0; k<num_faces; k++ ) {
93  unsigned size;
94  assert(fread(&size,1,sizeof(unsigned),fp));
95  faces[k].resize(size);
96  for( int f=0; f<size; ++f ) {
97  unsigned v;
98  assert(fread(&v,1,sizeof(unsigned),fp));
99  faces[k][f] = v;
100  }
101  }
102  }
103  //
104  fclose(fp);
105  return objects;
106  }
113  static void write_hacd ( std::string path, const std::vector<hacd_io::convex_object> &objects ) {
114  //
115  FILE *fp = fopen(path.c_str(),"wb");
116  assert(fp);
117  //
118  unsigned version (0x0001);
119  fwrite(&version,1,sizeof(unsigned),fp);
120  //
121  unsigned num_objects (objects.size());
122  fwrite(&num_objects,1,sizeof(unsigned),fp);
123  //
124  for( unsigned n=0; n<objects.size(); ++n ) {
125  //
126  const auto &obj = objects[n];
127  const auto &vertices = obj.vertices;
128  const auto &faces = obj.faces;
129  //
130  unsigned num_vertices = vertices.size();
131  unsigned num_faces = faces.size();
132  //
133  fwrite(&num_vertices,1,sizeof(unsigned),fp);
134  fwrite(&num_faces,1,sizeof(unsigned),fp);
135  //
136  for( size_t k=0; k<vertices.size(); k++ ) {
137  fwrite(&vertices[k],1,sizeof(vec3d),fp);
138  }
139  //
140  for( size_t k=0; k<faces.size(); k++ ) {
141  unsigned size = faces[k].size();
142  fwrite(&size,1,sizeof(unsigned),fp);
143  for( int f=0; f<size; ++f ) {
144  unsigned v (faces[k][f]);
145  fwrite(&v,1,sizeof(unsigned),fp);
146  }
147  }
148  }
149  //
150  fclose(fp);
151  }
